# Add Docker's official GPG key:
sudo apt update
sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# Add the repository to Apt sources:
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

# Docker Commands Reference Script

# --- Check if Docker is working fine ---
docker version

# --- Download an image ---
docker images
docker pull nginx:latest
docker images

# --- Start a container from the image ---
docker run --name test -d -p 8080:80 nginx:latest
# Open in browser: http://localhost:8080
docker ps

# --- Execute a command inside the container ---
docker exec -it test /bin/bash
ls
exit

# --- Delete the container ---
docker ps
docker stop test
docker rm test
docker ps -a

# --- Common base images and services ---
# Ubuntu (Base OS)
docker run -it ubuntu

# Nginx (Web server)
docker run -d -p 80:80 nginx

# MySQL (Database)
docker run -d -e MYSQL_ROOT_PASSWORD=1234 mysql

# MongoDB (NoSQL DB)
docker run -d -p 27017:27017 mongo

# Python (Runtime)
docker run -it python

# --- Cleanup: Deletes ALL stopped containers ---
docker container prune

#------------------- Q8. Shared Volume Between Containers -------------------

# Step 1: Create a Docker volume
docker volume create shared-data

# Step 2: Run first container with shared volume
docker run -it --name writer -v shared-data:/data alpine sh

# Inside writer container, write data
echo "Hello from Writer Container!" > /data/message.txt
exit

# Step 3: Run second container with same volume
docker run -it --name reader -v shared-data:/data alpine sh

# Inside reader container, read data
cat /data/message.txt
exit

# Expected Output:
# Hello from Writer Container!

#------------------- Q3. Docker Compose: MySQL with Volume -------------------

# Step 1: Create docker-compose.yml
vim docker-compose.yml

version: '3'
services:
  mysql:
    image: mysql:8
    container_name: my-mysql
    environment:
      MYSQL_ROOT_PASSWORD: root123
      MYSQL_DATABASE: cdac
    ports:
      - "3306:3306"
    volumes:
      - mysql-data:/var/lib/mysql

volumes:
  mysql-data:
# -------------------------------------------------

# Step 2: Start container
docker-compose up -d

# Step 3: Enter MySQL inside container
docker exec -it my-mysql mysql -uroot -p

# Step 4: Run SQL commands
CREATE TABLE test (id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(100));
INSERT INTO test (name) VALUES ('Amit');
SELECT * FROM test;
exit;

# Step 5: Restart container
docker-compose down
docker-compose up -d

# Step 6: Verify data persists
docker exec -it my-mysql mysql -uroot -p
USE cdac;
SELECT * FROM test;



#------------------- Q4. Docker Compose: One Volume with 3 Nginx Containers -------------------

# Goal:
# Run three Nginx containers (nginx1, nginx2, nginx3) that share a single volume
# Each container serves different HTML content on different ports

# Step 1: Create a Docker volume
docker volume create shared-data

# Step 2: Create docker-compose.yml
vim docker-compose.yml

version: '3'
services:
  nginx1:
    image: nginx:alpine
    container_name: nginx1
    ports:
      - "8081:80"
    volumes:
      - shared-data:/usr/share/nginx/html
    command: /bin/sh -c "echo 'Hello from Container 1' > /usr/share/nginx/html/index.html && nginx -g 'daemon off;'"

  nginx2:
    image: nginx:alpine
    container_name: nginx2
    ports:
      - "8082:80"
    volumes:
      - shared-data:/usr/share/nginx/html
    command: /bin/sh -c "echo 'Hello from Container 2' > /usr/share/nginx/html/index.html && nginx -g 'daemon off;'"

  nginx3:
    image: nginx:alpine
    container_name: nginx3
    ports:
      - "8083:80"
    volumes:
      - shared-data:/usr/share/nginx/html
    command: /bin/sh -c "echo 'Hello from Container 3' > /usr/share/nginx/html/index.html && nginx -g 'daemon off;'"

volumes:
  shared-data:
# -------------------------------------------------

# Step 3: Run Docker Compose
docker-compose up -d

# Step 4: Verify output in browser
# Visit each container on its port:
# http://localhost:8081 → Hello from Container 1
# http://localhost:8082 → Hello from Container 2
# http://localhost:8083 → Hello from Container 3

# Step 5: Clean up (optional)
docker-compose down
docker volume rm shared-data
