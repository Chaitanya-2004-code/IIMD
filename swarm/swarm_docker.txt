#------------------- Docker Swarm with NGINX -------------------

# Step 1: Create project folder and HTML file
mkdir nginx_project
cd nginx_project
echo "<h1>Welcome to my custom NGINX server</h1>" > index.html

# Step 2: Create Dockerfile
# -------------------------------------------------
FROM nginx:latest
COPY index.html /usr/share/nginx/html/index.html
# -------------------------------------------------

# Step 3: Build and run image locally
docker build -t newdelthis/custom-nginx:1.0 .
docker run -d -p 9999:80 newdelthis/custom-nginx:1.0

# Verify in browser: http://localhost:9999

# Step 4: Push image to DockerHub
docker login
docker push newdelthis/custom-nginx:1.0

# Step 5: Initialize Docker Swarm
docker swarm init

# Step 6: Create service with 1 replica
docker service create --name nginx-service --replicas 1 -p 9999:80 newdelthis/custom-nginx:1.0
docker service ls

# Step 7: Scale service to 10 replicas
docker service scale nginx-service=10
docker service ps nginx-service

# Step 8: Scale service down to 2 replicas
docker service scale nginx-service=2
docker service ps nginx-service

# Step 9: Update image with new HTML
echo "<h1>Welcome to the updated NGINX server</h1>" > index.html
docker build -t newdelthis/custom-nginx:2.0 .
docker push newdelthis/custom-nginx:2.0

# Step 10: Update service to use new image
docker service update --image newdelthis/custom-nginx:2.0 nginx-service
docker service ps nginx-service

# Verify in browser: http://localhost:9999

# Step 11: Remove service (cleanup)
docker service rm nginx-service

#------------------- Q2. Docker Swarm: Python Worker Service -------------------

# Step 1: Create Python script
vim app.py
import time
while True:
    print("Running in Swarm...")
    time.sleep(5)

# Step 2: Create Dockerfile
# ----------------
vim Dockerfile

FROM python:3.11-slim
WORKDIR /app
COPY app.py .
CMD ["python", "app.py"]
# -------------------------------------------------

# Step 3: Build Docker image
docker build -t python-worker-app .

# Step 4: Deploy service in Swarm
docker swarm init
docker service create --name python-service --replicas 3 python-worker-app

# Step 5: Check logs from different containers
docker service ps python-service
docker logs <container_id>

# Expected Output (repeated every 5 seconds):
# Running in Swarm...

# Step 6: Verify task distribution
docker service ps python-service


#------------------- Q3. Docker Swarm: Scaling Service -------------------

# Step 1: Deploy Nginx service with 2 replicas
docker service create --name nginx-service --replicas 2 -p 8080:80 nginx

# Step 2: Scale service to 5 replicas
docker service scale nginx-service=5

# Step 3: Verify replicas
docker service ls
docker service ps nginx-service

# Step 4: Scale service back to 1 replica
docker service scale nginx-service=1

# Step 5: Observe container changes
docker service ps nginx-service
# (Extra replicas will be terminated, only 1 remains active)


#------------------- Q4. Docker Swarm: Rolling Update -------------------

# Step 1: Deploy service using nginx:1.25 image
docker service create --name rolling-nginx -p 8080:80 nginx:1.25

# Step 2: Update service to nginx:latest with rolling update options
docker service update \
  --image nginx:latest \
  --update-parallelism 1 \
  --update-delay 10s \
  rolling-nginx

# Step 3: Verify update status
docker service ps rolling-nginx

# Step 4: Roll back service if needed
docker service update --rollback rolling-nginx

#------------------- Q5. Docker Swarm: Remove Service and Leave Swarm -------------------

# Step 1: List all running services

# Step 3: Remove python-service
